import sys
import os

# arquivo ``financeiro-backend/main.py``.
# --------------------------------------------------------------
project_root = os.path.abspath(os.path.join(os.path.dirname(__file__), '..'))
backend_path = os.path.join(project_root, 'financeiro-backend')
sys.path.insert(0, backend_path)
